#### Function 1: `sign(payload, secret, alg = 'HS256')`
```typescript
import { SignJWT } from 'jose'

export async function sign(
  payload: JwtPayload,
  secret: string | Buffer,
  alg = 'HS256'
): Promise<string> {
  const encoder = new TextEncoder()
  const secretKey = typeof secret === 'string' ? encoder.encode(secret) : secret

  return new SignJWT(payload)
    .setProtectedHeader({ alg })
    .setIssuedAt()
    .sign(secretKey)
}
```

**Key Points:**
- Default algorithm: HS256
- Support string/Buffer secrets
- Return JWT string
- Error handling: Convert jose errors to clear messages

#### Function 2: `verify(token, secret, alg = 'HS256')`
```typescript
import { jwtVerify } from 'jose'

export async function verify(
  token: string,
  secret: string | Buffer,
  alg = 'HS256'
): Promise<JwtPayload> {
  const encoder = new TextEncoder()
  const secretKey = typeof secret === 'string' ? encoder.encode(secret) : secret

  try {
    const { payload } = await jwtVerify(token, secretKey)
    return payload as JwtPayload
  } catch (error) {
    // Convert jose errors to clear messages
    throw new Error(`JWT verification failed: ${error.message}`)
  }
}
```

**Key Points:**
- Validates signature and expiration
- Clear error messages for invalid/expired tokens
- Support string/Buffer secrets
- Return verified payload

#### Function 3: `decode(token)`
```typescript
import { decodeProtectedHeader } from 'jose'

export function decode(token: string): { header: JwtHeader; payload: any } {
  try {
    const [headerB64, payloadB64] = token.split('.')
    const header = JSON.parse(
      new TextDecoder().decode(Base64Url.toArrayBuffer(headerB64))
    )
    const payload = JSON.parse(
      new TextDecoder().decode(Base64Url.toArrayBuffer(payloadB64))
    )
    return { header, payload }
  } catch (error) {
    throw new Error(`JWT decode failed: ${error.message}`)
  }
}
```

**Key Points:**
- No verification, just decoding
- Handle malformed tokens gracefully
- Return { header, payload }

#### Function 4: `jwt(options: JwtOptions): GravitoMiddleware`
```typescript
export function jwt(options: JwtOptions): GravitoMiddleware {
  return async (ctx, next) => {
    const token = ctx.req.header('Authorization')?.replace('Bearer ', '')
      || ctx.req.cookie(options.cookie || 'jwt')

    if (!token) {
      return ctx.json({ error: 'Unauthorized' }, 401)
    }

    try {
      const payload = await verify(token, options.secret, options.alg)
      ctx.set('jwtPayload', payload)
    } catch (error) {
      return ctx.json({ error: 'Unauthorized' }, 401)
    }

    await next()
  }
}
```

**Key Points:**
- Middleware factory function
- Extract token from Authorization header or cookie
- Verify and store payload in context
- Return 401 on invalid token
- Support GravitoContext
