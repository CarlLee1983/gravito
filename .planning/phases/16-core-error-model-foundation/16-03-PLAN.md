1. **Create packages/core/tests/contract/helpers.ts** (per RESEARCH Pattern 4):
```typescript
import { expect } from 'bun:test'
import { GravitoException } from '../../src/exceptions/GravitoException'
import { InfrastructureException } from '../../src/exceptions/InfrastructureException'
import { DomainException } from '../../src/exceptions/DomainException'
import { SystemException } from '../../src/exceptions/SystemException'

export interface ContractAssertOptions {
  expectedCode: string
  expectedStatus: number
  expectedInstanceOf?: Function[]
  expectRetryable?: boolean
  expectCause?: boolean
}

export function assertGravitoException(
  err: unknown,
  opts: ContractAssertOptions
): void {
  // 1. Base instanceof check
  expect(err).toBeInstanceOf(Error)
  expect(err).toBeInstanceOf(GravitoException)

  const e = err as GravitoException

  // 2. Required structural fields
  expect(e.code).toBe(opts.expectedCode)
  expect(e.status).toBe(opts.expectedStatus)
  expect(typeof e.code).toBe('string')
  expect(typeof e.status).toBe('number')

  // 3. Additional instanceof checks (intermediate layers)
  if (opts.expectedInstanceOf) {
    for (const cls of opts.expectedInstanceOf) {
      expect(err).toBeInstanceOf(cls)
    }
  }

  // 4. Cause chain preservation (ERRM-03)
  if (opts.expectCause) {
    expect(e.cause).toBeDefined()
  }

  // 5. Retryable field for InfrastructureException (D-02)
  if (opts.expectRetryable !== undefined) {
    expect(err).toBeInstanceOf(InfrastructureException)
    expect((err as InfrastructureException).retryable).toBe(opts.expectRetryable)
  }

  // 6. Anti-regression: name field is set (not generic 'Error')
  expect(e.name).not.toBe('Error')
}
```

2. **Create packages/core/tests/contract/core-exceptions.contract.test.ts**:

Write contract tests covering these scenarios:

**a) HttpException contract:**
- Create `new HttpException(404, { message: 'Not found' })`
- Assert: instanceof GravitoException, code 'HTTP_ERROR', status 404
- Assert: NOT instanceof DomainException, SystemException, InfrastructureException

**b) AuthenticationException contract:**
- Create `new AuthenticationException()`
- Assert: instanceof GravitoException AND instanceof DomainException
- Assert: code 'UNAUTHENTICATED', status 401

**c) AuthorizationException contract:**
- Create `new AuthorizationException()`
- Assert: instanceof GravitoException AND instanceof DomainException
- Assert: code 'FORBIDDEN', status 403

**d) ValidationException contract:**
- Create `new ValidationException([{ field: 'email', message: 'required' }])`
- Assert: instanceof GravitoException AND instanceof DomainException
- Assert: code 'VALIDATION_ERROR', status 422
- Assert: errors array has length 1
- Assert: withRedirect and withInput return `this` (chainable)

**e) CircularDependencyException contract:**
- Create `new CircularDependencyException('B', ['A'])`
- Assert: instanceof GravitoException AND instanceof SystemException
- Assert: code 'system.circular_dependency', status 500
- Assert: message contains 'A -> B'

**f) ConfigurationException contract:**
- Create `new ConfigurationException('Missing DB_URL')`
- Assert: instanceof GravitoException AND instanceof SystemException
- Assert: code 'system.configuration_error', status 500

**g) ModelNotFoundException contract:**
- Create `new ModelNotFoundException('User', 42)`
- Assert: instanceof GravitoException
- Assert: code 'NOT_FOUND', status 404

**h) Cause chain preservation (ERRM-03):**
- Create original: `new TypeError('connection refused')`
- Create wrapper: `new HttpException(500, { cause: original, message: 'Service error' })`
- Assert: wrapper.cause === original
- Assert: wrapper.cause instanceof TypeError

**i) Cross-layer cause chain:**
- Create original: `new Error('ECONNREFUSED')`
- Create `new ConfigurationException('DB config invalid', { cause: original })`
- Assert: cause chain preserved through SystemException -> GravitoException

**j) Object.setPrototypeOf verification (D-04):**
For each concrete exception class, verify that:
- `err instanceof ConcreteClass` is true
- `err instanceof IntermediateLayer` is true
- `err instanceof GravitoException` is true
- `err instanceof Error` is true

Use `describe('GravitoException Contract Tests', () => { ... })` as the top-level describe block with nested `describe` blocks for each category (hierarchy, fields, cause chain, instanceof).

CRITICAL: Tests MUST NOT assert on `.message` strings (except CircularDependencyException's path format which is structural). Tests assert `.code`, `.status`, `instanceof` only — per research anti-pattern guidance.
