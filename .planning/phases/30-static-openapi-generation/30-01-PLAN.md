3. **Replace the entire `convertToOpenApi` function** (lines 12-39) with:

```typescript
/**
 * Convert Zod schema to JSON Schema for OpenAPI 3.1 embedding.
 * Uses Zod v4 native z.toJSONSchema() — no external dependency.
 * Returns {} for null/undefined/non-Zod inputs.
 * Throws on conversion failure per D-07.
 */
function convertToOpenApi(schema: unknown): Record<string, unknown> {
  if (!schema || typeof schema !== 'object' || !('_def' in schema)) {
    return {}
  }
  try {
    const result = z.toJSONSchema(schema as z.ZodType) as Record<string, unknown>
    // Remove $schema — OpenAPI 3.1 embeds schemas inline, not as standalone documents
    const { $schema, ...rest } = result
    if (Object.keys(rest).length === 0) {
      throw new Error('z.toJSONSchema returned empty result')
    }
    return rest
  } catch (err) {
    throw new Error(`Zod schema conversion failed: ${err instanceof Error ? err.message : String(err)}`)
  }
}
```

For each of the three schema conversion sites (`schema.body`, `schema.params`, `schema.query`, `schema.response`), wrap the `convertToOpenApi()` call so that if it throws, a `CliError` is raised with the route's method and path for diagnostics. The outer `catch` block at line 186 already calls `process.exit(1)`, so this satisfies D-05. Example for body:

```typescript
if (schema.body) {
  try {
    const bodySchema = convertToOpenApi(schema.body)
```
