    ```typescript
    import {
      wrap,
      circuitBreaker as cockatielCB,
      ConsecutiveBreaker,
      retry,
      handleAll,
      handleWhen,
      ExponentialBackoff,
      timeout as cockatielTimeout,
      TimeoutStrategy,
      BrokenCircuitError,
    } from 'cockatiel'
    import { InfrastructureException } from '@gravito/core'
    import { CircuitOpenException } from '../exceptions/CircuitOpenException'
    import { RetryExhaustedException } from '../exceptions/RetryExhaustedException'
    import type { ResiliencePolicy } from './ResiliencePolicy'

    // Internal registry for named circuit breakers (shared across calls)
    const cbRegistry = new Map<string, ReturnType<typeof cockatielCB>>()

    export async function withResilience<T>(
      fn: () => Promise<T>,
      policy: ResiliencePolicy,
    ): Promise<T> {
      const policies: any[] = []

      // 1. Timeout (outermost)
      if (policy.timeout != null) {
        policies.push(cockatielTimeout(policy.timeout, TimeoutStrategy.Cooperative))
      }

      // 2. Circuit Breaker (middle — wraps retry) per D-08
      if (policy.circuitBreaker != null) {
        const cbOpts = typeof policy.circuitBreaker === 'string'
          ? { name: policy.circuitBreaker }
          : policy.circuitBreaker
        const cbKey = cbOpts.name
        if (!cbRegistry.has(cbKey)) {
          cbRegistry.set(cbKey, cockatielCB(handleAll, {
            halfOpenAfter: cbOpts.resetTimeout ?? 30_000,
            breaker: new ConsecutiveBreaker(cbOpts.failureThreshold ?? 5),
          }))
        }
        policies.push(cbRegistry.get(cbKey)!)
      }

      // 3. Retry (innermost — wraps fn directly) per D-08
      if (policy.retry != null) {
        const retryOpts = policy.retry
        if ((retryOpts as Record<string, unknown>).idempotent !== true) {
          throw new Error('withResilience: retry.idempotent must be true')
        }
        const handler = handleWhen((err: unknown) => {
          if (err instanceof InfrastructureException && err.retryable) return true
          if (retryOpts.retryOn) return retryOpts.retryOn(err)
          return false
        })
        policies.push(retry(handler, {
          maxAttempts: retryOpts.maxAttempts ?? 3,
          backoff: new ExponentialBackoff({
            initialDelay: retryOpts.baseDelayMs ?? 200,
            maxDelay: retryOpts.maxDelayMs ?? 30_000,
          }),
        }))
      }

      if (policies.length === 0) {
        return fn()
      }

      const combined = policies.length === 1 ? policies[0] : wrap(...policies)

      try {
        return await combined.execute(fn)
      } catch (err) {
        // Map cockatiel BrokenCircuitError -> Gravito CircuitOpenException
        if (err instanceof BrokenCircuitError) {
          throw new CircuitOpenException({ cause: err })
        }
        throw err
      }
    }

    /**
     * Clear the internal CB registry. For testing only.
     */
    export function _resetCBRegistry(): void {
      cbRegistry.clear()
    }
    ```

    **IMPORTANT per D-11:** The NODE_ENV=test behavior (throw on circuit-open rather than returning fallback) is already the default behavior of withResilience — it always throws CircuitOpenException when CB is open. The "fallback" path is for Phase 20's OrbitDegradationManager which will wrap withResilience. So D-11 is satisfied by design: withResilience always throws, never returns fallback.
